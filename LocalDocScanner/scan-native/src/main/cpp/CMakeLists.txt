cmake_minimum_required(VERSION 3.22)
project(localdoc_dewarp LANGUAGES C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROG OFF CACHE BOOL "" FORCE)
foreach(feature ZLIB PNG GIF JPEG TIFF WEBP OPENJPEG)
    set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(CMAKE_C_VISIBILITY_PRESET hidden)
add_subdirectory(../../../third_party/leptonica ${CMAKE_CURRENT_BINARY_DIR}/leptonica)
add_library(scanner_dewarp SHARED book_dewarp.c)
target_include_directories(scanner_dewarp PRIVATE
    ../../../third_party/leptonica/src ${CMAKE_CURRENT_BINARY_DIR}/leptonica/src)
target_link_libraries(scanner_dewarp PRIVATE leptonica jnigraphics log m)
target_link_options(scanner_dewarp PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
