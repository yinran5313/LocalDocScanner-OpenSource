cmake_minimum_required(VERSION 3.22)
project(shiyespm LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
get_filename_component(LOCALDOC_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${LOCALDOC_SOURCE_ROOT}=/localdoc" "-ffile-prefix-map=${CMAKE_ANDROID_NDK}=/android/ndk")
add_subdirectory(vendor)
add_library(shiyespm SHARED tokenizer.cpp)
target_include_directories(shiyespm PRIVATE vendor/src)
target_link_libraries(shiyespm sentencepiece-static log)
target_link_options(shiyespm PRIVATE "-Wl,-z,max-page-size=16384")
